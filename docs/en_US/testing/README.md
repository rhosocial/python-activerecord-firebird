# Testing

## Overview

This section covers testing principles and configuration for the Firebird backend. Firebird requires specific testing considerations due to its transaction handling and concurrency characteristics.

## Testing Principles

### Serial Execution Requirement

**CRITICAL**: Tests MUST run serially. Do NOT use `pytest -n auto` or parallel execution.

Firebird uses multi-version concurrency control (MVCC) and has specific transaction isolation characteristics that can cause test interference when run in parallel.

```bash
# Correct: Serial execution
PYTHONPATH=src pytest tests/

# Wrong: Parallel execution (will cause failures)
PYTHONPATH=src pytest -n auto tests/
```

### Test Isolation

Each test should be independent and clean up after itself:

```python
import pytest
from rhosocial.activerecord.backend.impl.firebird import FirebirdBackend

@pytest.fixture
def backend():
    """Create a fresh backend for each test."""
    config = FirebirdConnectionConfig(
        host="localhost",
        database="/tmp/test.fdb",
        username="SYSDBA",
        password="masterkey"
    )
    backend = FirebirdBackend(config=config)
    backend.connect()
    yield backend
    backend.disconnect()

def test_user_creation(backend):
    """Test creating a user."""
    backend.execute("CREATE TABLE test_users (id INTEGER, name VARCHAR(50))")
    backend.execute("INSERT INTO test_users (id, name) VALUES (1, 'Alice')")
    
    result = backend.execute("SELECT * FROM test_users WHERE id = 1")
    assert result[0][1] == 'Alice'
    
    # Cleanup
    backend.execute("DROP TABLE test_users")
```

### Test Database Setup

```python
# Create test database
def setup_test_database(backend):
    """Create test tables."""
    backend.execute("""
        CREATE TABLE users (
            id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            name VARCHAR(100) NOT NULL,
            email VARCHAR(255) NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    
    backend.execute("""
        CREATE TABLE orders (
            id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            user_id INTEGER NOT NULL,
            amount DECIMAL(10,2) NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)

def teardown_test_database(backend):
    """Drop test tables."""
    backend.execute("DROP TABLE IF EXISTS orders")
    backend.execute("DROP TABLE IF EXISTS users")
```

## Local Testing

### Docker-based Testing

Use Docker to run a Firebird server for testing:

```yaml
# docker-compose.yml
version: '3.8'
services:
  firebird:
    image: jacobalberty/firebird:4.0
    environment:
      - FIREBIRD_DATABASE=test.fdb
      - FIREBIRD_USER=SYSDBA
      - FIREBIRD_PASSWORD=masterkey
    ports:
      - "3050:3050"
    volumes:
      - firebird-data:/var/lib/firebird/4.0/data

volumes:
  firebird-data:
```

```bash
# Start Firebird server
docker-compose up -d

# Run tests
PYTHONPATH=src pytest tests/

# Stop server
docker-compose down
```

## Test Categories

### Basic CRUD Tests

```python
def test_create_user(backend):
    """Test creating a new user."""
    backend.execute("INSERT INTO users (name, email) VALUES (?, ?)", 
                    ("Alice", "alice@example.com"))
    result = backend.execute("SELECT * FROM users WHERE name = 'Alice'")
    assert len(result) == 1

def test_read_user(backend):
    """Test reading user data."""
    result = backend.execute("SELECT * FROM users WHERE id = 1")
    assert result[0][1] == "Alice"

def test_update_user(backend):
    """Test updating user data."""
    backend.execute("UPDATE users SET name = ? WHERE id = ?", ("Alice Smith", 1))
    result = backend.execute("SELECT * FROM users WHERE id = 1")
    assert result[0][1] == "Alice Smith"

def test_delete_user(backend):
    """Test deleting user."""
    backend.execute("DELETE FROM users WHERE id = 1")
    result = backend.execute("SELECT * FROM users WHERE id = 1")
    assert len(result) == 0
```

### Transaction Tests

```python
def test_transaction_commit(backend):
    """Test transaction commit."""
    backend.transaction.begin()
    backend.execute("INSERT INTO users (name, email) VALUES (?, ?)", 
                    ("Bob", "bob@example.com"))
    backend.transaction.commit()
    
    result = backend.execute("SELECT * FROM users WHERE name = 'Bob'")
    assert len(result) == 1

def test_transaction_rollback(backend):
    """Test transaction rollback."""
    backend.transaction.begin()
    backend.execute("INSERT INTO users (name, email) VALUES (?, ?)", 
                    ("Charlie", "charlie@example.com"))
    backend.transaction.rollback()
    
    result = backend.execute("SELECT * FROM users WHERE name = 'Charlie'")
    assert len(result) == 0
```

### Firebird-Specific Tests

```python
def test_execute_block(backend):
    """Test EXECUTE BLOCK functionality."""
    backend.execute("""
        EXECUTE BLOCK AS
        BEGIN
            INSERT INTO users (name, email) VALUES ('Test User', 'test@example.com');
        END
    """)
    
    result = backend.execute("SELECT * FROM users WHERE name = 'Test User'")
    assert len(result) == 1

def test_blob_handling(backend):
    """Test BLOB type handling."""
    backend.execute("""
        CREATE TABLE test_blobs (
            id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            data BLOB SUB_TYPE 0
        )
    """)
    
    test_data = b"Hello, Firebird BLOBs!"
    backend.execute("INSERT INTO test_blobs (data) VALUES (?)", (test_data,))
    
    result = backend.execute("SELECT data FROM test_blobs WHERE id = 1")
    assert result[0][0] == test_data
    
    backend.execute("DROP TABLE test_blobs")
```

## Running Tests

### Complete Test Suite

```bash
# Run all tests
PYTHONPATH=src pytest tests/

# Run with verbose output
PYTHONPATH=src pytest -v tests/

# Run specific test file
PYTHONPATH=src pytest tests/test_basic_crud.py

# Run specific test
PYTHONPATH=src pytest tests/test_basic_crud.py::test_create_user
```

### Test Markers

```bash
# Run only basic tests
PYTHONPATH=src pytest -m basic tests/

# Run only transaction tests
PYTHONPATH=src pytest -m transaction tests/

# Run only Firebird-specific tests
PYTHONPATH=src pytest -m firebird tests/
```
